def embed_texts(texts, batch=100):
    from openai import OpenAI  # pip install openai

    client = OpenAI(api_key=os.environ["OPENAI_API_KEY"])
    vecs, tokens = [], 0
    for i in range(0, len(texts), batch):
        r = client.embeddings.create(model=EMBED_MODEL, input=texts[i:i + batch])
        vecs += [d.embedding for d in r.data]
        tokens += r.usage.total_tokens
    return np.array(vecs, dtype="float32"), tokens


def build_embeddings(refresh=False):
    """조각 196개 + 질문 7개를 한 번에 임베딩해 저장한다."""
    chunks, qs = load_chunks(), questions()
    if EMB.exists() and not refresh:
        return load_embeddings()
    t0 = time.time()
    cv, t1 = embed_texts([c["text"] for c in chunks])
    qv, t2 = embed_texts([q["question"] for q in qs])
    OUT.mkdir(exist_ok=True)
    np.savez(EMB, chunk_ids=np.array([c["chunk_id"] for c in chunks]), chunk_vecs=cv,
             q_ids=np.array([q["id"] for q in qs]), q_vecs=qv,
             info=np.array(json.dumps({"model": EMBED_MODEL, "tokens": t1 + t2, "seconds": round(time.time() - t0, 1)})))
    return load_embeddings()

def unit(v):
    """코사인 유사도를 내적으로 계산하려고 길이를 1로 맞춘다."""
    v = np.array(v, dtype="float32")
    return v / np.linalg.norm(v, axis=-1, keepdims=True)

🧭 이런 상황이라면 어떻게 하겠습니까?

정책 · FAQ 조각이 100만 개로 늘었습니다. 검색 시간은 20ms 안이어야 하고, Flat과 같은 결과가 95% 이상이어야 합니다.

🧭 판단의 근거 보기

B가 맞습니다. 이 탭에서 본 것처럼 nprobe가 작으면 놓치고, 크면 Flat과 같아집니다. 정답은 데이터마다 다르므로 재서 고릅니다: Flat 결과를 기준으로 ‘Flat과 같은 결과 비율’과 시간을 nprobe별로 표로 만들고, 95% 이상인 가장 작은 nprobe를 씁니다. 흔히 nlist는 조각 수의 제곱근 근처(100만 → 1,000 안팎)에서 시작합니다. A는 100만 개를 모두 비교해 시간 조건을 넘기기 쉽고, C는 정확도 조건을 장담할 수 없습니다.

for M in (4, 16, 32):
    for ef in (8, 32, 128):
        t0 = time.perf_counter()
        h = faiss.IndexHNSWFlat(X.shape[1], M, faiss.METRIC_INNER_PRODUCT)
        h.hnsw.efConstruction = 40

        build = (time.perf_counter() - t0) * 1000
        h.hnsw.efSearch = ef
        _, I = h.search(Q, K)
        same = sum(len(set(I[i]) & set(F[i])) for i in range(len(Q)))

def policy_scores(i, cond=None):
    """정책마다 가장 가까운 조각의 점수 (cond가 있으면 조건에 맞는 조각만)"""
    s = X @ Q[i]

    return best

def top(scores, n, allow=None):
    keys = [d for d in scores if allow is None or d in allow]
    return sorted(keys, key=lambda d: -scores[d][0])[:n]


def expand(i):
    sc = policy_scores(i)
    seed = top(sc, a.seed)
    added = {x: d for d in seed for x in together(d) if x not in seed}
    cand = set(seed) | set(added)
    return sc, seed, added, top(sc, a.keep, cand)
  1. policy_scores: 정책마다 가장 가까운 조각의 점수를 그 정책의 점수로 씁니다. 같은 문서가 여러 칸을 차지하지 않습니다.
  2. together: 온톨로지 그래프에서 그 정책과 함께적용으로 이어진 정책을 찾습니다.
  3. expand: 상위 정책 5개(seed)와 그 이웃을 후보로 넓히고, 후보 안에서 벡터 점수 상위 7개(keep)를 남깁니다.

def cache_key(text, model=None):
    """캐시 열쇠 = 모델 이름 + 글 내용의 해시. 글이 한 글자라도 바뀌거나 모델이 바뀌면 다른 열쇠가 된다."""
    return hashlib.sha256(f"{model or EMBED_MODEL}\n{text}".encode("utf-8")).hexdigest()[:20]

def embed_with_cache(texts, apply=False, model=None):
    """캐시에 있으면 꺼내 쓰고, 없는 것만 API로 임베딩한다. apply=False면 무엇을 임베딩할지만 알려 준다."""
    cache = load_cache()
    keys = [cache_key(t, model) for t in texts]
    miss = [i for i, k in enumerate(keys) if k not in cache]
    tokens = 0
    if miss and apply:
        vecs, tokens = embed_texts([texts[i] for i in miss])
        for i, v in zip(miss, vecs):
            cache[keys[i]] = v
        save_cache(cache)
    vecs = [cache.get(k) for k in keys]
    return vecs, {"total": len(texts), "hit": len(texts) - len(miss), "miss": miss, "tokens": tokens}
