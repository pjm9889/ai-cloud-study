def load_graph(path="outputs_m2/ontology.json"):
    """노드에는 종류(type)를, 선에는 관계 이름과 근거(doc_id, section, evidence)를 붙인다."""
    onto = json.loads((ROOT / path).read_text(encoding="utf-8"))
    G = nx.MultiDiGraph(schema_version=onto["schema_version"])
    for n in onto["nodes"]:
        G.add_node(n["id"], type=n["type"])
    for e in onto["edges"]:
        G.add_edge(e["subject"], e["object"], key=e["predicate"], predicate=e["predicate"],
                   doc_id=e["doc_id"], section=e["section"], evidence=e["evidence"])
    return G
  1. nx.MultiDiGraph(): 방향이 있고, 같은 두 점 사이에 선을 여러 개 둘 수 있는 그래프입니다.
  2. 노드마다 type(정책, 상품유형 …)을 붙입니다.
  3. 엣지마다 관계 이름과 근거(doc_id · section · evidence)를 붙입니다. 그래프를 따라가 찾은 결과를 원문 문장으로 되짚을 수 있게 하기 위해서입니다.

06. 실습 11 · 경로 찾기 — 단어가 겹치지 않는 문서에 닿기

두 노드 사이의 가장 짧은 경로를 찾아, 몇 홉을 어떤 관계로 건너는지 근거와 함께 봅니다.

두 노드 사이를 몇 번 건너야 닿는지 찾습니다. 평가 질문 Q-001(제주 VIP 냉장 케이크 배송 지연 · 변질)에 필요한 EXC-005(VIP 제주 추가배송비)는 ‘냉장’과 단어가 겹치지 않습니다.

코드 읽기 — 실행하기 전에

목적경로 찾기
→
입력출발 · 도착 노드
→

→

→

📄 m3_11_hops.py · --path (발췌) (실제 코드 발췌)
if a.path:
    s, t = a.path
    try:
        p = nx.shortest_path(U, s, t)
    except nx.NetworkXNoPath:
        raise SystemExit(f"{s}와 {t} 사이에 경로가 없습니다")
    print(f"{s} → {t} · {len(p) - 1}홉")
    for x, y in zip(p, p[1:]):
        k, d = next(iter(G.get_edge_data(x, y, default={}).items()), (None, None))
        if d is None:
            k, d = next(iter(G.get_edge_data(y, x).items()))
            print(f"  {x} ←{k}─ {y}   · {d['doc_id']} · {d['evidence'][:50]}")
        else:
            print(f"  {x} ─{k}→ {y}   · {d['doc_id']} · {d['evidence'][:50]}")
    raise SystemExit
  1. nx.shortest_path: 방향을 무시한 그래프에서 선을 가장 적게 건너는 경로를 찾습니다.
  2. 건넌 선마다 관계 이름과 근거 문장을 출력합니다. ‘←’는 선의 방향이 거꾸로라는 뜻입니다.

def hop_search(G, starts, hops=2):
    """1홉: 출발 개념에 닿은 정책 (점수 = 닿은 출발 개념 수). 2홉: 그 정책에서 '함께적용'으로 이어진 정책."""
    U = G.to_undirected(as_view=True)
    score, via = {}, {}
    for s in starts:
        for p in policies(G, U.neighbors(s)):
            score[p] = score.get(p, 0) + 1
            via.setdefault(p, []).append(s)
    hop1 = sorted(score, key=lambda p: (-score[p], p))
    hop2 = []
    if hops >= 2:
        for p in hop1:
            for a, b, k in list(G.out_edges(p, keys=True)) + list(G.in_edges(p, keys=True)):
                other = b if a == p else a
                if k == "함께적용" and other not in score and other not in hop2:
                    hop2.append(other)
                    via[other] = [f"{p} (함께적용)"]
    return hop1, hop2, score, via
  1. 1홉: 출발 노드에 바로 닿은 정책을 모읍니다. 점수는 그 정책에 닿은 출발 노드의 수입니다. 질문의 개념을 많이 다루는 정책일수록 점수가 높습니다.
  2. 2홉: 1홉 정책에서 함께적용 선만 따라 한 번 더 건넙니다. 모든 선을 따라가면 거의 모든 정책이 모이기 때문에, 관계를 골라 건넙니다.
