from dotenv import load_dotenv
load_dotenv()

import pandas as pd
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser

llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)
parser = StrOutputParser()

docs = pd.read_csv("travel_docs.csv")      # doc_id, category, source, title, body

# 문서 번호 하나로 자료 가져오기
def get_doc(doc_id: str) -> str:
    r = docs[docs["doc_id"] == doc_id].iloc[0]
    return f"[{r['doc_id']}] {r['title']}\n{r['body']}"

# 문서 번호 여러 개를 한 번에
def get_docs(doc_ids: list) -> str:
    return "\n\n".join(get_doc(i) for i in doc_ids)

print(docs.shape)

1MessagesPlaceholder · RunnableWithMessageHistory

손님이 앞에서 말한 일정·인원·조건을 기억하는 여행 상담 챗봇을 만듭니다. 노트북 [1] 오즈 내과 상담 코드와 같은 구조입니다.

1-1. 이전 대화를 기억하는 상담

MessagesPlaceholder로 지난 대화가 들어갈 자리를 만들고, RunnableWithMessageHistory로 손님(session_id)별 대화를 저장합니다.

from langchain_core.prompts import ChatPromptTemplate, MessagesPlaceholder
from langchain_core.runnables.history import RunnableWithMessageHistory
from langchain_community.chat_message_histories import ChatMessageHistory

# 프롬프트: history 자리에 이전 대화가 들어감
prompt = ChatPromptTemplate.from_messages([
    ("system", "당신은 바람길여행의 상담 AI입니다. 손님이 이전에 말한 일정, 인원, 조건을 기억하고 일관되게 응대하세요."),
    MessagesPlaceholder(variable_name="history"),   # 지난 대화가 주입되는 자리
    ("human", "{question}"),
])

chain = prompt | llm                 # 이미 아는 LCEL 파이프

store = {}                           # 세션ID(손님)별 대화 저장소 (데모용 메모리)

def get_history(session_id: str):
    if session_id not in store:
        store[session_id] = ChatMessageHistory()
    return store[session_id]

chat = RunnableWithMessageHistory(
    chain,
    get_history,
    input_messages_key="question",    # 입력 중 무엇이 손님 발화인지
    history_messages_key="history",   # 프롬프트의 어느 자리에 넣을지 (위 placeholder와 일치)
)

# session_id(손님 구분)만 넘기면 대화 이력이 알아서 쌓이고 주입된다
r1 = chat.invoke(
    {"question": "6월에 부모님 모시고 셋이서 2박3일로 솔바람섬에 가려고 해요."},
    config={"configurable": {"session_id": "guest_kim"}},
)
print("AI:", r1.content)

r2 = chat.invoke(
    {"question": "제가 아까 언제, 몇 명이 간다고 했죠?"},
    config={"configurable": {"session_id": "guest_kim"}},
)
print("AI:", r2.content)
AI: 6월에 부모님과 함께 세 분이 2박 3일로 솔바람섬에 가시는군요! 숙소, 이동 방법, 일정 중 어떤 것부터 도와드릴까요?
AI: 6월에 부모님과 함께 총 3명이 2박 3일 일정으로 솔바람섬에 가신다고 하셨습니다.
설명
  • 두 번째 질문에는 날짜도 인원도 없습니다. 그래도 답할 수 있는 것은 첫 번째 대화가 history 자리에 자동으로 들어갔기 때문입니다.
  • input_messages_key는 invoke에 넣는 dict의 key(question)와, history_messages_key는 MessagesPlaceholder의 이름(history)과 같아야 합니다.
  • chain = prompt | llm처럼 파서를 붙이지 않았습니다. 대화 기록에는 AI 메시지가 저장되므로 노트북과 같이 .content로 꺼내 출력합니다.

2RunnablePassthrough

규정 자료(context)와 손님 질문(question)을 함께 프롬프트에 넣을 때, 질문을 그대로 보존하는 방법입니다. 노트북 [2]의 진료기록 예제와 같은 구조입니다.

2-1. 에러: question 자리까지 조회 함수가 채워 버림

dict의 두 자리 모두에 조회 함수를 연결하면, 손님이 실제로 물은 질문이 프롬프트에서 사라집니다.

# 규정 DB 조회를 흉내 낸 함수 (실무에서는 상품 ID로 DB 조회)
def get_policy(question: str) -> str:
    return get_doc("T15")      # 바람길여행 취소·환불 규정


rag_prompt = ChatPromptTemplate.from_template(
    """다음 여행사 규정을 근거로 손님 질문에 답하세요.

[규정]
{context}

[손님 질문]
{question}"""
)

#  문제 코드 — question 자리에도 조회 함수를 그대로 연결
#    원본 질문이 사라지고, [손님 질문] 자리에까지 규정이 들어감
rag_chain = (
    {
        "context":  get_policy,
        "question": get_policy,   # 여기가 문제: 원본 질문이 보존되지 않음
    }
    | rag_prompt
    | llm
    | parser
)

print(rag_chain.invoke("출발 3일 전에 취소하면 얼마 돌려받아요?"))
print("=" * 60)
print(rag_chain.invoke("강아지"))

2-3. 해결: question 자리에 RunnablePassthrough()

rag_chain = (

    | rag_prompt
    | llm
    | parser
)

print(rag_chain.invoke("출발 3일 전에 취소하면 얼마 돌려받아요?"))

3-2. 해결: RunnableParallel로 문자열 하나 → dict

RunnableParallel이 같은 입력을 받아 두 개의 값을 동시에 만들고, dict로 묶어 프롬프트에 넘깁니다.

travel_parallel = RunnableParallel({
    "request": lambda x: x.strip(),          # 손님 입력 가공 (앞뒤 공백 제거)

4RunnableBranch · RunnableSequence

질문 종류(환불 / 숙소 / 인사)에 따라 다른 체인으로 보냅니다. 노트북 [4] 병원 챗봇(예약 / 증상 / 인사)과 같은 구조입니다.

4-1. 문제: 하나의 프롬프트로 모든 질문을 처리할 때

4-2. RunnableBranch로 의도별 분기

① 전용 체인 3개를 만들고 ② 질문의 의도를 분류한 뒤 ③ RunnableBranch로 맞는 체인에 보냅니다.

from langchain_core.runnables import RunnableBranch

# ① 분기될 전용 체인 3개 ─────────────────────────
refund_prompt = ChatPromptTemplate.from_template(
    """당신은 바람길여행 환불 담당자입니다. 아래 규정을 근거로 답하세요.

[취소·환불 규정]
{policy}

[문의]
{question}"""
)
refund_chain = (
    {
        "policy": lambda x: get_doc("T15"),
        "question": lambda x: x["question"],
    }
    | refund_prompt | llm | parser
)

stay_prompt = ChatPromptTemplate.from_template(
    """당신은 은하군 숙소 안내 AI입니다. 아래 숙소 자료를 근거로 답하세요.

[숙소 자료]
{stays}

[문의]
{question}"""
)
stay_chain = (
    {
        "stays": lambda x: get_docs(["T06", "T07", "T08"]),
        "question": lambda x: x["question"],
    }
    | stay_prompt | llm | parser
)

general_prompt = ChatPromptTemplate.from_template(
    "당신은 여행사 안내 AI입니다. 다음 인사에 짧고 친절하게 답하세요.\n\n{question}"
)
general_chain = general_prompt | llm | parser

# ② 의도 분류기 — intent 키를 붙여줌 ──────────────
def classify_intent(x: dict) -> dict:
    q = x["question"]
    if any(w in q for w in ["취소", "환불"]):
        intent = "환불"
    elif any(w in q for w in ["숙소", "리조트", "펜션", "호텔", "묵"]):
        intent = "숙소"
    else:
        intent = "일반"
    return {"question": q, "intent": intent}

# ③ RunnableBranch — 의도에 따라 분기 ─────────────
branch_router = RunnableBranch(
    (lambda x: x["intent"] == "환불", refund_chain),   # 조건 참이면 환불 체인
    (lambda x: x["intent"] == "숙소", stay_chain),     # 조건 참이면 숙소 체인
    general_chain,                                     # 어디에도 안 걸리면 기본 체인
)

# ④ 분류 → 분기 연결
final_chain = classify_intent | branch_router

# 테스트
print(final_chain.invoke({"question": "출발 3일 전에 취소하면 환불돼요?"}))
print("=" * 60)
print(final_chain.invoke({"question": "강아지랑 같이 묵을 수 있는 숙소 있어요?"}))
print("=" * 60)
print(final_chain.invoke({"question": "안녕하세요"}))

============================================================

============================================================

설명
  • RunnableBranch((조건, 체인), (조건, 체인), 기본체인): 위에서부터 조건을 검사해서 처음으로 참인 체인을 실행합니다. 모두 거짓이면 마지막 기본 체인을 실행합니다.
  • 각 체인은 자기에게 필요한 자료만 넣습니다. 환불 체인은 T15, 숙소 체인은 T06~T08, 인사 체인은 자료 없음.
  • classify_intent는 일반 함수인데 |로 연결할 수 있습니다. 랭체인이 자동으로 RunnableLambda로 바꿔 줍니다.
  • 분류기는 키워드만 봅니다. 키워드로 의도를 잘못 읽는 경우는 종합 탭 9에서 검증합니다.

chatbot = classify_intent | branch_router   # 이미 RunnableSequence

# ① invoke — 한 번에 실행
print(chatbot.invoke({"question": "출발 7일 전에 취소하면요?"}))
print()

# ② stream — 답변을 토큰 단위로 실시간 출력 (챗봇 UI에 적합)
for token in chatbot.stream({"question": "리조트 체크인 시간이 언제예요?"}):
    print(token, end="", flush=True)
print("\n")

# ③ batch — 여러 손님 메시지를 한 번에 처리
results = chatbot.batch([

    {"question": "안녕하세요"},
])
for r in results:
    print("-", r)

6-3. 벡터DB 구축: 임베딩 후 FAISS 저장

문장을 '의미를 담은 숫자(임베딩)'로 바꿔 저장합니다. 그래서 "강아지"로 물어도 "반려동물"이 적힌 자료를 찾을 수 있습니다. 자세한 내용은 다음 시간에 배우고, 지금은 "의미로 찾아 주는 검색기"로만 이해하면 됩니다.

from langchain_openai import OpenAIEmbeddings
from langchain_community.vectorstores import FAISS

# 임베딩 모델: 문장 → 의미 벡터로 변환하는 도구
embeddings = OpenAIEmbeddings(model="text-embedding-3-small")

# FAISS.from_documents: 모든 Document의 page_content를 임베딩해 벡터DB에 저장
vectorstore = FAISS.from_documents(documents, embeddings)

# as_retriever: 벡터DB를 '검색기'로 변환. k=3 → 의미가 가까운 상위 3건
retriever = vectorstore.as_retriever(search_kwargs={"k": 3})

# 검색된 Document 리스트 → 프롬프트에 넣을 문자열 하나
def format_docs(docs):
    return "\n\n".join(d.page_content for d in docs)

print("벡터DB 구축 완료. 벡터 수:", vectorstore.index.ntotal)
벡터DB 구축 완료. 벡터 수: 22
설명
  • retriever는 질문을 넣으면 Document 3개를 돌려줍니다. 뒤의 RAG 체인에서 context를 채우는 데 그대로 씁니다.
  • format_docs는 Document 리스트를 프롬프트에 넣을 문자열 하나로 합칩니다. retriever | format_docs로 이어 씁니다.

7종합② RAG 체인과 통계 체인

7-1. RAG 체인: RunnableParallel + RunnablePassthrough

탭 2의 get_policy 자리를 retriever | format_docs로 바꾼 것입니다. 이제 질문마다 맞는 자료를 검색해서 넣습니다.

from langchain_core.prompts import ChatPromptTemplate
from langchain_core.runnables import RunnablePassthrough

    """당신은 은하군 여행 상담 AI입니다.

{context}

[손님 질문]
{question}"""
)

    {

    | llm
    | parser
)

  • "context": retriever | format_docs: 질문 → 검색(Document 3개) → 문자열. "question": RunnablePassthrough(): 질문 그대로. 이 두 줄이 RAG 체인의 핵심입니다.

7-2. 통계 체인: RunnableLambda로 집계 함수 넣기

"어떤 분류의 자료가 제일 많아?" 같은 질문은 검색(3건)으로는 답할 수 없습니다. 전체 22건을 집계해야 합니다. 일반 파이썬 함수를 RunnableLambda로 감싸 체인에 넣습니다.

from collections import Counter
from langchain_core.runnables import RunnableLambda

# CSV(records)를 직접 집계하는 함수
def doc_stats(_input: str) -> str:
    cat = Counter(r["category"] for r in records)
    src = Counter(r["source"] for r in records)
    lines = ["[분류별 자료 수]"] + [f"- {k}: {v}건" for k, v in cat.most_common()]
    lines += ["", "[출처별 자료 수]"] + [f"- {k}: {v}건" for k, v in src.most_common()]
    lines += ["", f"[전체] {len(records)}건"]
    return "\n".join(lines)

stats_prompt = ChatPromptTemplate.from_template(
    """아래 통계를 바탕으로 질문에 간결히 답하세요.

[통계]
{stats}

[질문]
{question}"""
)

stats_chain = (
    {
        "stats": RunnableLambda(doc_stats),   # CSV 집계 함수 삽입
        "question": RunnablePassthrough(),
    }
    | stats_prompt
    | llm
    | parser
)

print(doc_stats(""))          # 함수 결과 먼저 확인

8-1. 검색 결과를 근거로 손님 요청 정리

from pydantic import BaseModel, Field
from typing import List
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.runnables import RunnablePassthrough

class TripCheck(BaseModel):
    main_needs: List[str] = Field(description="손님 요청에서 뽑은 핵심 조건")
    related_places: List[str] = Field(description="검색된 자료에 나온 장소·상품 이름들")
    best_option: str = Field(description="자료를 근거로 가장 알맞은 장소·상품 하나")
    caution: str = Field(description="주의할 점 한 줄")
    advice: str = Field(description="손님에게 줄 한 줄 안내")

triage_llm = llm.with_structured_output(TripCheck)

    """아래 [검색된 자료]를 참고하여 손님 요청을 정리하세요.

[검색된 자료]
{context}

[손님 요청]
{question}"""
)

# 검색(context) + 원본(question) → 구조화 LLM
triage_chain = (
    {
        "context": retriever | format_docs,   # CSV 검색을 근거로 사용
        "question": RunnablePassthrough(),

    | triage_prompt

)

  • Field(description=...)는 모델에게 "이 칸에는 무엇을 넣으라"는 설명입니다.
  • 관련 장소에 '별빛전망대', '파도소리 리조트'가 들어 있습니다. 검색된 자료에 정말 있었을까요? 8-2에서 확인합니다.

9종합④ RunnableBranch: 의도별 분기 라우팅

질문을 통계 / 여행 상담 / 인사로 나눠 각기 다른 체인으로 보냅니다. 통계면 집계 체인(탭 7-2), 여행 상담이면 RAG 체인(탭 7-1), 인사면 검색 없는 짧은 체인입니다. (탭 6 실행 후)

9-1. 분류 → 분기 라우터

탭 7의 RAG 체인과 통계 체인을 그대로 가져와서 분기로 연결합니다. 이 탭만 실행해도 되도록 두 체인 코드를 모두 넣었습니다.

# ── 탭 7-1 RAG 체인 ─────────────────────────
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.runnables import RunnablePassthrough

rag_prompt = ChatPromptTemplate.from_template(
    """당신은 은하군 여행 상담 AI입니다.
아래 [검색된 자료]를 근거로 손님 질문에 답하세요.
자료에 없는 내용은 지어내지 말고 '자료에 없습니다'라고 답하세요.

[검색된 자료]
{context}

[손님 질문]
{question}"""
)

rag_chain = (
    {
        "context": retriever | format_docs,   # CSV 검색 → 문자열
        "question": RunnablePassthrough(),     # 원본 질문 보존
    }
    | rag_prompt
    | llm
    | parser
)

# ── 탭 7-2 통계 체인 ────────────────────────
from collections import Counter
from langchain_core.runnables import RunnableLambda

# CSV(records)를 직접 집계하는 함수
def doc_stats(_input: str) -> str:
    cat = Counter(r["category"] for r in records)
    src = Counter(r["source"] for r in records)
    lines = ["[분류별 자료 수]"] + [f"- {k}: {v}건" for k, v in cat.most_common()]
    lines += ["", "[출처별 자료 수]"] + [f"- {k}: {v}건" for k, v in src.most_common()]
    lines += ["", f"[전체] {len(records)}건"]
    return "\n".join(lines)

stats_prompt = ChatPromptTemplate.from_template(
    """아래 통계를 바탕으로 질문에 간결히 답하세요.

[통계]
{stats}

[질문]
{question}"""
)

stats_chain = (
    {
        "stats": RunnableLambda(doc_stats),   # CSV 집계 함수 삽입
        "question": RunnablePassthrough(),
    }
    | stats_prompt
    | llm
    | parser
)

# ── 분기 ──────────────────────────────────
from langchain_core.runnables import RunnableBranch

# 일반 인사용 짧은 체인 (검색 불필요)
general_chain = (
    ChatPromptTemplate.from_template(
        "당신은 은하군 여행 안내 AI입니다. 다음에 짧고 친절히 답하세요.\n\n{question}"
    )
    | llm | parser
)

# 의도 분류 키워드
STAT_WORDS = ["몇", "통계", "분포", "가장", "제일", "비율", "건수"]
TRAVEL_WORDS = ["여행", "숙소", "펜션", "리조트", "호텔", "식당", "맛집", "페리", "배",
                "환불", "취소", "예약", "전망대", "축제", "가격", "가이드", "강아지", "반려동물"]

# 질문 → {question, intent} 로 의도 판별
def classify(x):
    q = x if isinstance(x, str) else x["question"]
    if any(w in q for w in STAT_WORDS):
        intent = "stats"
    elif any(w in q for w in TRAVEL_WORDS):
        intent = "travel"
    else:
        intent = "general"
    return {"question": q, "intent": intent}

# 의도에 따라 CSV 기반 체인으로 분기
branch = RunnableBranch(
    (lambda x: x["intent"] == "stats",  lambda x: stats_chain.invoke(x["question"])),  # CSV 집계
    (lambda x: x["intent"] == "travel", lambda x: rag_chain.invoke(x["question"])),    # CSV 검색
    lambda x: general_chain.invoke({"question": x["question"]}),                       # 일반(검색 없음)
)

# 분류 → 분기. | 가 곧 RunnableSequence
router_chain = RunnableLambda(classify) | branch

print("● 통계:", router_chain.invoke("정책 자료는 몇 건이야?"))
print("● 상담:", router_chain.invoke("반딧불 축제 야간 탐방은 예약해야 해?"))
print("● 일반:", router_chain.invoke("안녕하세요"))
● 통계: 정책 자료는 3건입니다.
● 상담: 네, 반딧불 축제 기간의 야간 탐방은 사전 예약제로 운영됩니다. 탐방 중에는 손전등과 휴대폰 불빛 사용이 금지됩니다.
● 일반: 안녕하세요! 은하군 여행에 대해 궁금한 점이 있으면 편하게 물어보세요.
설명
  • 분기 조건 뒤에 체인 대신 lambda x: stats_chain.invoke(x["question"])를 넣었습니다. 분기로 들어오는 값은 {question, intent} dict인데, 통계·RAG 체인은 질문 문자열을 받기 때문에 꺼내서 넘겨 준 것입니다.
  • 통계·RAG·인사 체인은 각자 데이터를 다르게 씁니다(집계 / 검색 / 사용 안 함). 분기가 질문 종류에 맞는 방식을 골라 줍니다.

10종합⑤ 멀티턴 RAG

탭 1의 대화 기억과 탭 7의 검색을 한 체인에 넣습니다. 손님이 말한 조건을 기억하면서, 질문마다 자료를 검색해 근거로 답합니다. (탭 6 실행 후)

10-1. 대화 기억 + 검색

프롬프트에 {context}(검색 자료), history(지난 대화), {question}(지금 질문) 세 자리를 만듭니다.

from langchain_core.prompts import ChatPromptTemplate, MessagesPlaceholder
from langchain_core.runnables.history import RunnableWithMessageHistory
from langchain_community.chat_message_histories import ChatMessageHistory

chat_prompt = ChatPromptTemplate.from_messages([
    ("system",
     "당신은 은하군 여행 상담 AI입니다. 아래 [검색된 자료]와 이전 대화를 함께 참고해 "
     "일관되게 응대하되, 자료에 없는 내용은 지어내지 마세요.\n\n"
     "[검색된 자료]\n{context}"),
    MessagesPlaceholder(variable_name="history"),   # 지난 대화가 들어갈 자리
    ("human", "{question}"),
])

# 입력 dict의 question으로 CSV를 검색해 context를 채우고, 원본 질문·이력은 그대로 통과
chat_core = (
    {
        "context": (lambda x: x["question"]) | retriever | format_docs,  # CSV 검색
        "question": (lambda x: x["question"]),                            # 원본 질문 보존
        "history": (lambda x: x.get("history", [])),                      # 이력 통과
    }
    | chat_prompt
    | llm
    | parser
)

store = {}  # 세션ID별 대화 저장소 (데모용 메모리)
def get_history(session_id: str):
    if session_id not in store:
        store[session_id] = ChatMessageHistory()
    return store[session_id]

chat = RunnableWithMessageHistory(
    chat_core, get_history,
    input_messages_key="question",     # {question} 과 일치해야 함
    history_messages_key="history",    # MessagesPlaceholder 와 일치해야 함
)

cfg = {"configurable": {"session_id": "guestA"}}
print("Q1:", chat.invoke({"question": "6월에 강아지 데리고 청연계곡 쪽으로 여행 가려고 해요."}, config=cfg))
print()
print("Q2:", chat.invoke({"question": "강아지랑 같이 할 수 있는 거 정리해 주세요."}, config=cfg))
Q1: 6월에 청연계곡으로 가신다면 계곡쉼 펜션을 추천드립니다. 청연계곡 입구에 있고 반려동물 동반 투숙이 가능하며, 객실마다 개별 바비큐 공간이 있습니다. 또 6월에는 반딧불숲에서 반딧불 축제가 열립니다.

Q2: 강아지와 함께라면 청연계곡 입구의 계곡쉼 펜션에서 숙박하실 수 있습니다. 반려동물 동반 투숙이 가능하고 객실마다 바비큐 공간이 있어 편하게 지내실 수 있습니다. 주말에는 예약이 빨리 마감되니 미리 예약하세요.
설명
  • chat_core 맨 앞 dict는 탭 7의 RAG 체인과 같은 RunnableParallel입니다. 다른 점은 입력이 문자열이 아니라 dict({question, history})라서 lambda x: x["question"]로 질문을 꺼내 검색한다는 것입니다.
  • Q2에는 '청연계곡'이라는 말이 없지만, Q1 대화가 history로 들어가서 청연계곡 이야기로 이어 답했습니다.

10-2. 검증: Q2는 무엇을 검색했고, 무엇을 빠뜨렸나

Q2 질문으로 실제 검색된 자료를 보고, 답에서 빠진 내용이 없는지 확인합니다.

# Q2가 실제로 검색한 자료 확인 — Q2 질문 그대로 retriever에 넣어봄
q2 = "강아지랑 같이 할 수 있는 거 정리해 주세요."
hits = retriever.invoke(q2)

print("=== Q2 질문으로 검색된 자료 ===")
for i, d in enumerate(hits, 1):
    print(f"[{i}] {d.metadata['doc_id']} | {d.metadata['title']}")

# 반려동물 관련 내용이 자료마다 어떻게 다른지 직접 확인
print("\n=== 자료별 반려동물 규정 ===")
for r in records:
    if "반려동물" in r["body"]:
        sentence = [s for s in r["body"].split(". ") if "반려동물" in s and ("가능" in s or "없" in s)][0]
        print(f"  {r['doc_id']} {r['title']}: {sentence}")
=== Q2 질문으로 검색된 자료 ===
[1] T07 | 계곡쉼 펜션
[2] T18 | [FAQ] 바람길여행 자주 묻는 질문
[3] T21 | [혼동 주의] 별빛캠핑장

=== 자료별 반려동물 규정 ===
  T07 계곡쉼 펜션: 계곡쉼 펜션의 객실은 8개이며 반려동물 동반 투숙이 가능하다
  T18 [FAQ] 바람길여행 자주 묻는 질문: Q. 투어에 반려동물을 데려갈 수 있나요? A. 바람길여행 투어 상품에는 반려동물을 동반할 수 없습니다
  T21 [혼동 주의] 별빛캠핑장: 별빛캠핑장은 반려동물 동반이 불가능하다
설명
  • 지어낸 내용(환각)은 없습니다. 답에 나온 계곡쉼 펜션 정보는 T07과 일치합니다.
  • 그런데 빠진 내용이 있습니다. 검색된 T18에는 "바람길여행 투어 상품(예: 청연계곡 반딧불 트레킹)에는 반려동물을 데려갈 수 없다"가 있는데 답에는 없습니다. 모델이 여러 자료 중 "가능"한 쪽만 골라 말한 선택적 인용입니다.
  • 강아지와 반딧불 트레킹을 예약하려던 손님에게는 중요한 정보가 빠진 것입니다. 10-3에서 프롬프트로 고칩니다.

10-3. 개선: CoT + 퓨샷으로 선택적 인용 잡기

① 퓨샷: "자료마다 답이 다르면 모두 제시"하는 모범 답변 1개를 보여 주고 ② CoT: "자료 확인 → 다른 점 점검 → 답변" 순서로 생각하게 합니다.

# ─────────────────────────────────────────────────────────────
# 문제: LLM이 검색된 여러 자료 중 일부만 골라 말함 (반려동물 '불가' 자료 누락)
# 해결: ① 퓨샷으로 "자료마다 답이 다르면 모두 제시"하는 답변 형식을 보여 줌
#       ② CoT로 "먼저 자료를 전부 확인 → 다른지 점검 → 답변" 순서를 강제
# ─────────────────────────────────────────────────────────────
from langchain_core.prompts import ChatPromptTemplate, MessagesPlaceholder
from langchain_core.runnables.history import RunnableWithMessageHistory
from langchain_community.chat_message_histories import ChatMessageHistory

# 퓨샷 예시: 자료마다 답이 다를 때 어떻게 답해야 하는지 모범 답변 1개
fewshot_example = """[좋은 답변 예시]
질문: 강아지 데려가도 돼요?
검색된 자료: T06(파도소리 리조트: 반려동물 언급 없음), T18(바람길여행 투어: 반려동물 동반 불가)
답변: 자료마다 다릅니다. 바람길여행 투어 상품(T18)은 반려동물을 데려갈 수 없습니다.
파도소리 리조트(T06)는 자료에 반려동물 관련 내용이 없어 확인이 필요합니다."""

# 시스템 프롬프트: 제약(환각 방지) + CoT(단계 강제) + 퓨샷(형식)
cot_system = (
    "당신은 은하군 여행 상담 AI입니다. 아래 [검색된 자료]에 나온 내용만 근거로 답하고, "
    "자료에 없는 장소나 규정은 절대 만들지 마세요.\n\n"
    "반드시 다음 순서로 생각하여 답하세요.\n"
    "1단계: 검색된 자료를 문서 번호·이름·내용까지 빠짐없이 확인한다.\n"
    "2단계: 같은 질문에 대해 자료마다 답(가능/불가 등)이 다른지 점검한다.\n"
    "3단계: 종합해 답하되, 답이 갈리면 일부만 고르지 말고 모두 밝힌다.\n"
    "'[혼동 주의]' 자료는 은하군과 관계없는 곳이므로 추천하지 말고 필요하면 구분만 해 준다.\n\n"
    f"{fewshot_example}\n\n"
    "[검색된 자료]\n{context}"
)

cot_prompt = ChatPromptTemplate.from_messages([
    ("system", cot_system),
    MessagesPlaceholder(variable_name="history"),
    ("human", "{question}"),
])

# 검색(context) + 원본질문(question) + 이력(history) 결합
cot_chat_core = (
    {
        "context": (lambda x: x["question"]) | retriever | format_docs,  # CSV 검색
        "question": (lambda x: x["question"]),                            # 원본 보존
        "history": (lambda x: x.get("history", [])),                      # 이력 통과
    }
    | cot_prompt
    | llm
    | parser
)

# 멀티턴 래핑 (10-1과 별도 저장소 사용)
cot_store = {}
def get_cot_history(session_id: str):
    if session_id not in cot_store:
        cot_store[session_id] = ChatMessageHistory()
    return cot_store[session_id]

cot_chat = RunnableWithMessageHistory(
    cot_chat_core, get_cot_history,
    input_messages_key="question",
    history_messages_key="history",
)

# 같은 대화 흐름으로 실행
cfg = {"configurable": {"session_id": "guestB"}}
print("Q1:", cot_chat.invoke({"question": "6월에 강아지 데리고 청연계곡 쪽으로 여행 가려고 해요."}, config=cfg))
print()
ans2 = cot_chat.invoke({"question": "강아지랑 같이 할 수 있는 거 정리해 주세요."}, config=cfg)
print("Q2:", ans2)
Q1: 청연계곡 입구의 계곡쉼 펜션(T07)은 반려동물 동반 투숙이 가능합니다. 다만 바람길여행 투어 상품(T18)은 반려동물을 데려갈 수 없으니, 반딧불 트레킹 같은 투어는 강아지와 함께 참여할 수 없습니다.

Q2: 자료마다 다릅니다.
- 가능: 계곡쉼 펜션(T07) 숙박. 반려동물 동반 투숙이 가능하고 객실마다 바비큐 공간이 있습니다.
- 불가: 바람길여행 투어 상품(T18). 청연계곡 반딧불 트레킹 등 투어에는 강아지를 데려갈 수 없습니다.
참고로 별빛캠핑장(T21)은 은하군이 아닌 해밀군의 캠핑장이며, 반려동물 동반도 불가능합니다.
설명
  • 이제 '가능'과 '불가'를 함께 밝혔습니다. 2단계(자료마다 답이 다른지 점검)를 시킨 효과입니다.
  • 퓨샷 예시는 형식만 보여 주는 용도입니다. 예시에 실제 질문과 같은 자료(T07)를 쓰지 않은 이유는, 모델이 예시 내용을 답에 베끼지 않게 하기 위해서입니다.
  • cot_system은 f-string이 아닌 문자열과 f-string을 이어 붙였습니다. {context}는 f-string이 아닌 부분에 있어서 프롬프트 빈칸으로 남습니다.
정리 선택적 인용은 프롬프트(CoT + 퓨샷)로 잡았습니다. 남은 문제(질문과 관련 낮은 자료가 검색에 섞임, 대화 맥락을 반영하지 못한 검색)는 다음 단계인 실전 RAG에서 재순위화, 관련도 점수 필터링, 검색 질의 다시 쓰기(query rewriting)로 다룹니다.

11종합⑥ 안정성: with_retry · with_fallbacks · stream

탭 7의 RAG 체인에 실패 대비 장치를 붙입니다. 검색 로직은 그대로 두고 LLM 부분만 바꿉니다. (탭 6 실행 후)

11-1. 재시도 + 대체 모델

with_retry: 일시적 오류면 자동 재시도. with_fallbacks: 그래도 실패하면 대체 모델로 우회.

from langchain_core.prompts import ChatPromptTemplate
from langchain_core.runnables import RunnablePassthrough

rag_prompt = ChatPromptTemplate.from_template(
    """당신은 은하군 여행 상담 AI입니다.
아래 [검색된 자료]를 근거로 손님 질문에 답하세요.
자료에 없는 내용은 지어내지 말고 '자료에 없습니다'라고 답하세요.

[검색된 자료]
{context}

[손님 질문]
{question}"""
)
from langchain_openai import ChatOpenAI

# 주력 모델 + 재시도 + 대체 모델을 결합한 견고한 LLM
backup_llm = ChatOpenAI(model="gpt-3.5-turbo", temperature=0)

robust_llm = (
    llm.with_retry(stop_after_attempt=2)      # 먼저 2회까지 재시도
       .with_fallbacks([backup_llm])          # 그래도 실패하면 대체 모델
)

robust_rag = (
    {"context": retriever | format_docs, "question": RunnablePassthrough()}
    | rag_prompt
    | robust_llm
    | parser
)

print(robust_rag.invoke("파도소리 리조트 예약을 체크인 5일 전에 취소하면 환불돼요?"))
파도소리 리조트는 해온호스피탈리티가 운영하며, 체크인 3일에서 6일 전에 취소하면 50%를 환불받을 수 있습니다.
설명
  • 체인 구조는 탭 7-1과 같고, llm 자리만 robust_llm으로 바뀌었습니다.
  • 정상일 때는 주력 모델(gpt-4o-mini)이 답합니다. 주력 모델이 계속 실패할 때만 backup_llm이 대신 답합니다.

11-2. stream: 실시간 출력

같은 체인을 stream으로 실행하면 답이 만들어지는 대로 글자가 흘러나옵니다.

# stream: 생성되는 즉시 토큰을 흘려보내 실시간 출력 (ChatGPT 처럼)
print("스트리밍 출력 ↓")
for chunk in robust_rag.stream("솔바람섬에서 렌터카 빌릴 수 있어요? 섬 안에서는 어떻게 다녀요?"):
    print(chunk, end="", flush=True)
print()
스트리밍 출력 ↓
솔바람섬에는 렌터카가 없고 전기 자전거 대여소만 있습니다. 섬 둘레길은 천천히 걸으면 약 세 시간이 걸리니, 전기 자전거나 도보로 다니시면 됩니다.
설명
  • robust_rag는 retry·fallback이 붙어 있어도 stream을 그대로 쓸 수 있습니다.
  • 챗봇 화면에서는 대부분 stream으로 보여 줍니다. 답이 길어도 첫 글자가 바로 나와서 기다리는 느낌이 줄어듭니다.
