03. FAISS 벡터를 OpenSearch로 옮기기(01) — reconstruct

FAISS 파일 안의 벡터 6개를 index.reconstruct(i)로 하나씩 꺼내, 원문과 함께 OpenSearch 문서 6건으로 저장합니다. 임베딩 API는 한 번도 부르지 않습니다.

📍 어디서 터미널 · vector_service.py를 옆에 열어 두고✅ 먼저 01탭 점검 통과
rag_faiss/index.faiss ── reconstruct(0..5) ──→ 1536차원 벡터 6개 ─┐
rag_faiss/chunks.json ── chunks[0..5] ───────→ 원문 6개 ──────────┤
                                                                   ↓
                                    OpenSearch rag_docs  문서 6건
                                    { chunk_id, title, content, embedding }
vector_service.py · 인덱스 만들기와 옮기기 (줄바꿈만 정리)
def create_vector_index(c, dimension, recreate=False):
    if c.indices.exists(index=RAG_INDEX):
        if recreate: c.indices.delete(index=RAG_INDEX)     # 실습용: 있으면 지우고 새로
        else: return False
    c.indices.create(index=RAG_INDEX, body={
        "settings": {"index": {"knn": True}},
        "mappings": {"properties": {
            "chunk_id": {"type": "integer"},
            "title":    {"type": "keyword"},
            "content":  {"type": "text"},
            "embedding": {"type": "knn_vector", "dimension": dimension,
                          "method": {"name": "hnsw", "space_type": "cosinesimil", "engine": "faiss"}},
        }},
    })
    return True

def transfer_faiss_vectors(c, index, chunks):
    for i, chunk in enumerate(chunks):
        c.index(index=RAG_INDEX, id=i, body={
            "chunk_id": i,
            "title": chunk.splitlines()[0].replace("## ", ""),   # 첫 줄 "## 서울 지역 현황" → "서울 지역 현황"
            "content": chunk,
            "embedding": index.reconstruct(i).tolist(),          # FAISS에서 i번 벡터 꺼내기
        })
    c.indices.refresh(index=RAG_INDEX)

reconstruct(i)는 “FAISS야, i번째에 넣은 벡터를 다시 보여 줘”입니다. M4에서 IndexFlatIP(전부 그대로 저장하는 인덱스)를 썼기 때문에 원래 숫자가 그대로 나옵니다. 벡터를 압축하는 인덱스(PQ 등)였다면 근사값만 나옵니다. 그리고 FAISS의 0번 = chunks.json의 0번이라는 순서 약속 덕분에 벡터와 원문을 짝지을 수 있습니다.

06. OpenSearch k-NN으로 검색(03) — index.search 대신 knn 쿼리

질문만 새로 임베딩하고, 검색은 OpenSearch가 합니다. M4의 search_faiss()가 하던 일을 knn_search()가 대신합니다.

📍 어디서 터미널✅ 먼저 03탭
vector_service.py · knn_search (줄바꿈만 정리)
def knn_search(ai, c, q, k=3):
    v = ai.embeddings.create(model=EMBEDDING_MODEL, input=q).data[0].embedding   # 질문만 임베딩
    return c.search(index=RAG_INDEX, body={
        "size": k,
        "_source": ["chunk_id", "title", "content"],      # embedding은 받지 않음
        "query": {"knn": {"embedding": {"vector": v, "k": k}}},
    })["hits"]["hits"]
03_opensearch_knn_search.py
from clients import get_openai_client, get_opensearch_client
from vector_service import knn_search

hits = knn_search(get_openai_client(), get_opensearch_client(), "광주 지역의 시장 특성은?", 3)
for n, h in enumerate(hits, 1):
    print(n, round(h["_score"], 4), h["_source"]["title"])

from clients import get_openai_client, get_opensearch_client
from config import RAG_INDEX, EMBEDDING_MODEL

ai, c = get_openai_client(), get_opensearch_client()

def embed(q):
    return ai.embeddings.create(model=EMBEDDING_MODEL, input=q).data[0].embedding

def knn_filtered(q, k=3, titles=None):
    knn = {"vector": embed(q), "k": k}
    if titles:
        knn["filter"] = {"terms": {"title": titles}}       # A4-M2 03탭 terms
    hits = c.search(index=RAG_INDEX, body={"size": k, "_source": ["title"],
                    "query": {"knn": {"embedding": knn}}})["hits"]["hits"]
    return [(round(h["_score"], 4), h["_source"]["title"]) for h in hits]

② Hybrid — 키워드와 벡터 섞기

m_vector_ext.py 에 이어서 · A4-M3 17탭의 pipeline 재사용
c.transport.perform_request("PUT", "/_search/pipeline/hybrid-pipe", body={
    "phase_results_processors": [{"normalization-processor": {
        "normalization": {"technique": "min_max"},
        "combination": {"technique": "arithmetic_mean", "parameters": {"weights": [0.3, 0.7]}},
    }}]})

def hybrid(q, k=3):
    body = {"size": k, "_source": ["title"], "query": {"hybrid": {"queries": [
        {"match": {"content": q}},                                  # 키워드 (content는 text)
        {"knn": {"embedding": {"vector": embed(q), "k": k}}},       # 벡터
    ]}}}
    hits = c.search(index=RAG_INDEX, body=body, params={"search_pipeline": "hybrid-pipe"})["hits"]["hits"]
    return [(round(h["_score"], 4), h["_source"]["title"]) for h in hits]

for q in ["프로젝터", "인구 940만 명인 곳"]:
    print(q, "| 벡터:", knn_filtered(q), "| Hybrid:", hybrid(q))

from clients import get_openai_client, get_opensearch_client
from vector_service import load_existing_faiss
from config import EMBEDDING_MODEL

INDEX = "rag_docs_v2"
c, ai = get_opensearch_client(), get_openai_client()
index, chunks = load_existing_faiss()

if c.indices.exists(index=INDEX):
    c.indices.delete(index=INDEX)
c.indices.create(index=INDEX, body={"settings": {"index": {"knn": True}}, "mappings": {"properties": {
    "title": {"type": "keyword"}, "region": {"type": "keyword"}, "content": {"type": "text"},
    "embedding": {"type": "knn_vector", "dimension": index.d,
                  "method": {"name": "hnsw", "space_type": "cosinesimil", "engine": "faiss"}}}}})

for i, chunk in enumerate(chunks):
    title = chunk.splitlines()[0].replace("## ", "")
    region = ______________________________          # "광주 지역 현황" → "광주", 그 외 → "전체"
    c.index(index=INDEX, id=i, body={"title": title, "region": region, "content": chunk,
                                     "embedding": index.reconstruct(i).tolist()})
c.indices.refresh(index=INDEX)

v = ai.embeddings.create(model=EMBEDDING_MODEL, input="주요 수요 제품은?").data[0].embedding
for reg in [None, "광주"]:
    knn = {"vector": v, "k": 3}
    if reg:
        knn["filter"] = {"term": {"region": reg}}
    hits = c.search(index=INDEX, body={"size": 3, "_source": ["title", "region"],
                    "query": {"knn": {"embedding": knn}}})["hits"]["hits"]
    print(reg or "필터 없음", [(h["_source"]["region"], h["_source"]["title"]) for h in hits])

from config import EMBEDDING_MODEL
RAG_INDEX = "rag_docs"

def count_docs(os_client):
    return os_client.count(index=RAG_INDEX)["count"]

def search_opensearch(openai_client, os_client, question, k=3):
    v = openai_client.embeddings.create(model=EMBEDDING_MODEL, input=question).data[0].embedding
    hits = os_client.search(index=RAG_INDEX, body={"size": k, "_source": ["title", "content"],
                            "query": {"knn": {"embedding": {"vector": v, "k": k}}}})["hits"]["hits"]
    # app.py가 기대하는 모양(rank, score, title, content)으로 맞춰 돌려줌
    return [{"rank": i + 1, "score": h["_score"], "title": "## " + h["_source"]["title"],
             "content": h["_source"]["content"]} for i, h in enumerate(hits)]
확인 포인트

핵심은 search_opensearch가 M4의 search_faiss와 같은 모양의 결과(rank, score, title, content)를 돌려준다는 점입니다. 그래서 화면 코드(hybrid_answer, 표, 원문 보기)는 하나도 안 고칩니다. 저장소를 바꿀 때 “돌려주는 모양을 맞추면 나머지는 그대로”라는 설계 감각을 익히는 미션입니다. Evidence 탭의 유사도 숫자가 M4보다 커진 것은 미션 3의 점수 변환 때문입니다.

Q. 그럼 FAISS는 이제 필요 없나요?

상황에 따라 다릅니다. 혼자 쓰는 작은 고정 문서, 서버 없이 빠르게 실험할 때는 FAISS 파일이 가장 간단합니다(M4). 여러 사람이 쓰고, 문서가 자주 바뀌고, 필터·키워드·운영 화면이 필요하면 OpenSearch입니다(M5). 그리고 OpenSearch 안에서도 FAISS가 엔진으로 일하고 있습니다.
